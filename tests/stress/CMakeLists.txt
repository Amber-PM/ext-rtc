cmake_minimum_required(VERSION 3.20)
project(rtc_stress_client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()
find_package(LibDataChannel 0.24.6 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(rtc-stress-client client.cpp)
target_compile_definitions(rtc-stress-client PRIVATE RTC_ENABLE_MEDIA=0 RTC_ENABLE_WEBSOCKET=0)
target_link_libraries(rtc-stress-client PRIVATE LibDataChannel::LibDataChannel Threads::Threads)
