cmake_minimum_required(VERSION 3.20)
project(rtc_transport_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LibDataChannel 0.24.6 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES LibDataChannel::LibDataChannel Threads::Threads)
if(WIN32)
	list(APPEND CMAKE_REQUIRED_LIBRARIES ws2_32)
endif()
set(CMAKE_REQUIRED_DEFINITIONS -DRTC_ENABLE_MEDIA=0 -DRTC_ENABLE_WEBSOCKET=0)
unset(RTC_BACKEND_MESSAGE_LIFETIME CACHE)
check_cxx_source_compiles("
#include <rtc/datachannel.hpp>
#include <rtc/ext_rtc_backend.h>
bool (rtc::DataChannel::*volatile sendOwner)(const rtc::byte*, size_t, std::shared_ptr<void>) = &rtc::DataChannel::send;
int main(){ return !sendOwner || !rtcExtMessageLifetimeIdentity(); }
" RTC_BACKEND_MESSAGE_LIFETIME)
if(NOT RTC_BACKEND_MESSAGE_LIFETIME)
	message(FATAL_ERROR "ext-rtc requires the pinned message-lifetime patched backend")
endif()
unset(CMAKE_REQUIRED_LIBRARIES)
unset(CMAKE_REQUIRED_DEFINITIONS)
add_library(rtc-test-core STATIC
	../../src/peer/peer.cpp ../../src/channel/channel.cpp ../../src/core/buffers.cpp
	../../src/channel/receive.cpp ../../src/channel/send.cpp ../../src/core/dispatcher.cpp)
target_compile_features(rtc-test-core PUBLIC cxx_std_20)
target_compile_definitions(rtc-test-core PUBLIC RTC_ENABLE_MEDIA=0 RTC_ENABLE_WEBSOCKET=0)
target_link_libraries(rtc-test-core PUBLIC LibDataChannel::LibDataChannel Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
	target_link_libraries(rtc-test-core PUBLIC ws2_32)
endif()
add_executable(rtc-offerer offerer.cpp)
target_link_libraries(rtc-offerer PRIVATE rtc-test-core)
add_library(rtc-windows-macros OBJECT windows-macros.cpp)
target_link_libraries(rtc-windows-macros PRIVATE rtc-test-core)
enable_testing()
add_executable(rtc-backend-identity backend-identity.cpp ../../src/backend/backend_info.cpp)
target_link_libraries(rtc-backend-identity PRIVATE rtc-test-core)
add_test(NAME rtc-backend-identity COMMAND rtc-backend-identity)
add_executable(rtc-portable-lifetime portable-lifetime.cpp)
target_link_libraries(rtc-portable-lifetime PRIVATE rtc-test-core)
add_test(NAME rtc-portable-lifetime COMMAND rtc-portable-lifetime)
set_tests_properties(rtc-portable-lifetime PROPERTIES TIMEOUT 45)
foreach(name lifecycle allocation channels receive send budgets concurrent)
	add_executable(rtc-${name} ${name}.cpp)
	target_link_libraries(rtc-${name} PRIVATE rtc-test-core)
	add_test(NAME rtc-${name} COMMAND rtc-${name})
	set_tests_properties(rtc-${name} PROPERTIES TIMEOUT 120)
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	add_executable(rtc-teardown teardown.cpp)
	target_link_libraries(rtc-teardown PRIVATE rtc-test-core)
	target_link_options(rtc-teardown PRIVATE "LINKER:--export-dynamic")
	foreach(mode backlog normal peers budget request concurrent construct-close)
		add_test(NAME rtc-teardown-${mode} COMMAND rtc-teardown ${mode})
		set_tests_properties(rtc-teardown-${mode} PROPERTIES TIMEOUT 45)
	endforeach()
	foreach(name original allocation)
		add_executable(rtc-teardown-${name} teardown-${name}.cpp)
		target_link_libraries(rtc-teardown-${name} PRIVATE rtc-test-core)
		target_link_options(rtc-teardown-${name} PRIVATE "LINKER:--export-dynamic")
		add_test(NAME rtc-teardown-${name} COMMAND rtc-teardown-${name} teardown)
		set_tests_properties(rtc-teardown-${name} PROPERTIES TIMEOUT 45)
	endforeach()
	target_link_options(rtc-send PRIVATE "LINKER:--export-dynamic")
	add_executable(rtc-credits credits.cpp)
	target_link_libraries(rtc-credits PRIVATE rtc-test-core)
	target_link_options(rtc-credits PRIVATE "LINKER:--export-dynamic")
	add_test(NAME rtc-credits COMMAND rtc-credits)
	set_tests_properties(rtc-credits PROPERTIES TIMEOUT 30)
	add_executable(rtc-credit-close credit-close.cpp)
	target_link_libraries(rtc-credit-close PRIVATE rtc-test-core)
	target_link_options(rtc-credit-close PRIVATE "LINKER:--export-dynamic")
	foreach(mode local remote both teardown during-send direct queued before-send unregister failure-close disconnect)
		add_test(NAME rtc-credit-close-${mode} COMMAND rtc-credit-close ${mode})
		set_tests_properties(rtc-credit-close-${mode} PROPERTIES TIMEOUT 30)
	endforeach()
	add_test(NAME rtc-backend-failure COMMAND rtc-send --backend-failure)
	add_test(NAME rtc-false-close COMMAND rtc-send --false-close)
	set_tests_properties(rtc-backend-failure rtc-false-close PROPERTIES TIMEOUT 30)
	add_executable(rtc-candidate-failure candidate-failure.cpp)
	target_link_libraries(rtc-candidate-failure PRIVATE rtc-test-core)
	target_link_options(rtc-candidate-failure PRIVATE "LINKER:--wrap=_ZNK3rtc9Candidate9candidateB5cxx11Ev")
	add_test(NAME rtc-candidate-failure COMMAND rtc-candidate-failure)
	set_tests_properties(rtc-candidate-failure PROPERTIES TIMEOUT 30)
endif()

add_executable(rtc-failures failures.cpp)
target_link_libraries(rtc-failures PRIVATE rtc-test-core)
add_test(NAME rtc-receive-budget COMMAND rtc-failures budget)
add_test(NAME rtc-remote-close COMMAND rtc-failures close)
set_tests_properties(rtc-receive-budget rtc-remote-close PROPERTIES TIMEOUT 30)
