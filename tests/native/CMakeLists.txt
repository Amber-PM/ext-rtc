cmake_minimum_required(VERSION 3.20)
project(rtc_transport_tests LANGUAGES CXX)
find_package(LibDataChannel 0.24.6 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(rtc-offerer offerer.cpp)
add_executable(rtc-lifecycle lifecycle.cpp ../../src/peer.cpp)
add_executable(rtc-allocation allocation.cpp ../../src/peer.cpp)
add_library(rtc-windows-macros OBJECT windows-macros.cpp)
foreach(target rtc-offerer rtc-lifecycle rtc-allocation rtc-windows-macros)
	target_compile_features(${target} PRIVATE cxx_std_20)
	target_compile_definitions(${target} PRIVATE RTC_ENABLE_MEDIA=0 RTC_ENABLE_WEBSOCKET=0)
	target_link_libraries(${target} PRIVATE LibDataChannel::LibDataChannel Threads::Threads ${CMAKE_DL_LIBS})
	if(WIN32)
		target_link_libraries(${target} PRIVATE ws2_32)
	endif()
endforeach()
enable_testing()
add_test(NAME rtc-lifecycle COMMAND rtc-lifecycle)
add_test(NAME rtc-allocation COMMAND rtc-allocation)
set_tests_properties(rtc-lifecycle PROPERTIES TIMEOUT 120)
set_tests_properties(rtc-allocation PROPERTIES TIMEOUT 30)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	add_executable(rtc-candidate-failure candidate-failure.cpp ../../src/peer.cpp)
	target_compile_features(rtc-candidate-failure PRIVATE cxx_std_20)
	target_compile_definitions(rtc-candidate-failure PRIVATE RTC_ENABLE_MEDIA=0 RTC_ENABLE_WEBSOCKET=0)
	target_link_libraries(rtc-candidate-failure PRIVATE LibDataChannel::LibDataChannel Threads::Threads ${CMAKE_DL_LIBS})
	target_link_options(rtc-candidate-failure PRIVATE "LINKER:--wrap=_ZNK3rtc9Candidate9candidateB5cxx11Ev")
	add_test(NAME rtc-candidate-failure COMMAND rtc-candidate-failure)
	set_tests_properties(rtc-candidate-failure PROPERTIES TIMEOUT 30)
endif()
